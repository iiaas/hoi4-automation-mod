# 科技本身已不在这里：由 tools/hoi4_nologistics.py 的每日钩子在引擎内遍历科技库自动解锁。
# 这里只剩科技之外的内容：开局一次性的 add_daily_mastery / auto_design，以及按年份写死的专项项目。
# 每个年份段各自一个永久 flag，防止专项项目的一次性奖励被按天重复触发。
auto_research = {
	if = {
		limit = { NOT = { has_country_flag = auto_research_pre1936_done } }
		set_country_flag = auto_research_pre1936_done
		add_daily_mastery = { amount = 1 days = 3650 }
		auto_design = yes
	}
	if = {
		limit = {
			date > 1937.1.1
			NOT = { has_country_flag = auto_research_1937_done }
		}
		complete_special_project_1937 = yes
		set_country_flag = auto_research_1937_done
	}
	if = {
		limit = {
			date > 1938.1.1
			NOT = { has_country_flag = auto_research_1938_done }
		}
		complete_special_project_1938 = yes
		set_country_flag = auto_research_1938_done
	}
	if = {
		limit = {
			date > 1940.1.1
			NOT = { has_country_flag = auto_research_1940_done }
		}
		complete_special_project_1940 = yes
		set_country_flag = auto_research_1940_done
	}
	if = {
		limit = {
			date > 1941.1.1
			NOT = { has_country_flag = auto_research_1941_done }
		}
		complete_special_project_1941 = yes
		set_country_flag = auto_research_1941_done
	}
	if = {
		limit = {
			date > 1942.1.1
			NOT = { has_country_flag = auto_research_1942_done }
		}
		complete_special_project_1942 = yes
		set_country_flag = auto_research_1942_done
	}
	if = {
		limit = {
			date > 1943.1.1
			NOT = { has_country_flag = auto_research_1943_done }
		}
		complete_special_project_1943 = yes
		set_country_flag = auto_research_1943_done
	}
	if = {
		limit = {
			date > 1944.1.1
			NOT = { has_country_flag = auto_research_1944_done }
		}
		complete_special_project_1944 = yes
		set_country_flag = auto_research_1944_done
	}
	if = {
		limit = {
			date > 1945.1.1
			NOT = { has_country_flag = auto_research_1945_done }
		}
		complete_special_project_1945 = yes
		set_country_flag = auto_research_1945_done
	}
}
