on_actions = {
	on_state_control_changed = {
		effect = {
			if = {
				limit = { is_ai = no }
				FROM.FROM = {
					set_resistance = 0
					set_compliance = 70
				}
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				auto_design = yes
				auto_build = yes
				auto_production_naval = yes
			}
		}
	}
}
