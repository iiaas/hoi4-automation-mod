on_actions = {
	on_army_leader_daily = {
		effect = {
			if = {
				limit = { FROM = { is_ai = no } }
				supply_units = 24
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = { is_ai = no }
				FROM.FROM = {
					set_resistance = 0
					set_compliance = 70
				}
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				auto_research = yes
				auto_build = yes
				auto_focus_shine = yes
				auto_import_resources = yes
			}
		}
	}
}
